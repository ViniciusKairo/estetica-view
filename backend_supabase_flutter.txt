BACKEND SUPABASE + FLUTTER TCC
Consolidação do que foi definido nesta conversa
============================================================

OBJETIVO
------------------------------------------------------------
Este documento consolida, em ordem lógica, a estrutura de backend
para o aplicativo Flutter com Supabase, contemplando:

1) Banco de dados
2) Enums
3) Tabelas
4) Funções helper
5) RLS e policies
6) Storage
7) Edge Functions
8) Secrets
9) Observações arquiteturais

Observação:
Este material cobre o backend definido até aqui.
Não inclui ainda a implementação Flutter de:
- upload criptografado
- download + descriptografia
- composição antes|depois

============================================================

-- =========================================================
-- EXTENSÕES NECESSÁRIAS
-- =========================================================
create extension if not exists pgcrypto;

-- =========================================================
-- ENUMS DO SISTEMA
-- =========================================================
do $$
begin
  if not exists (select 1 from pg_type where typname = 'user_role') then
    create type public.user_role as enum ('admin', 'medico', 'paciente');
  end if;
end $$;

do $$
begin
  if not exists (select 1 from pg_type where typname = 'procedimento_status') then
    create type public.procedimento_status as enum (
      'pendente_imagens',
      'em_andamento',
      'concluido',
      'cancelado'
    );
  end if;
end $$;

do $$
begin
  if not exists (select 1 from pg_type where typname = 'tipo_imagem') then
    create type public.tipo_imagem as enum ('antes', 'depois');
  end if;
end $$;

do $$
begin
  if not exists (select 1 from pg_type where typname = 'solicitacao_status') then
    create type public.solicitacao_status as enum (
      'pendente',
      'aprovada',
      'reprovada'
    );
  end if;
end $$;

-- =========================================================
-- FUNÇÃO GENÉRICA PARA updated_at
-- =========================================================
create or replace function public.set_updated_at()
returns trigger
language plpgsql
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

  p_entity_type text,
  p_entity_id text default null,
  p_description text default null,

as $$

    p_entity_id,
    p_description,
    coalesce(p_metadata, '{}'::jsonb)
  );

$$;

-- =========================================================
-- VALIDAÇÃO OPCIONAL DE CONSISTÊNCIA DE ACESSO
-- =========================================================
create or replace function public.validate_acesso_imagem_procedimento()
returns trigger
language plpgsql
as $$
begin
  if new.is_active = false and new.revoked_at is null then
    raise exception 'Ao revogar um acesso, revoked_at deve ser informado.';
  end if;

  if new.is_active = true and new.revoked_at is not null then
    raise exception 'Acesso ativo não pode possuir revoked_at preenchido.';
  end if;

  return new;
end;
$$;

create trigger trg_validate_acesso_imagem_procedimento
before insert or update on public.acessos_imagem_procedimento
for each row
execute function public.validate_acesso_imagem_procedimento();

-- =========================================================
-- VALIDAÇÃO OPCIONAL DE CONSISTÊNCIA DE SOLICITAÇÃO
-- =========================================================
create or replace function public.validate_solicitacao_imagem()
returns trigger
language plpgsql
as $$
begin
  if new.status in ('aprovada', 'reprovada') and new.reviewed_at is null then
    raise exception 'Solicitações aprovadas ou reprovadas devem ter reviewed_at preenchido.';
  end if;

  return new;
end;
$$;

create trigger trg_validate_solicitacao_imagem
before insert or update on public.solicitacoes_imagem
for each row
execute function public.validate_solicitacao_imagem();

create or replace function public.current_profile_id()
returns uuid
language sql
stable
as $$
  select auth.uid();
$$;

create or replace function public.current_role()
returns public.user_role
language sql
stable
as $$
  select p.role
  from public.profiles p
  where p.id = auth.uid()
  limit 1;
$$;

create or replace function public.is_admin()
returns boolean
language sql
stable
as $$
  select coalesce(public.current_role() = 'admin', false);
$$;

create or replace function public.is_medico()
returns boolean
language sql
stable
as $$
  select coalesce(public.current_role() = 'medico', false);
$$;

create or replace function public.is_paciente()
returns boolean
language sql
stable
as $$
  select coalesce(public.current_role() = 'paciente', false);
$$;

create or replace function public.current_medico_id()
returns uuid
language sql
stable
as $$
  select m.id
  from public.medicos m
  where m.profile_id = auth.uid()
  limit 1;
$$;

create or replace function public.current_paciente_id()
returns uuid
language sql
stable
as $$
  select p.id
  from public.pacientes p
  where p.profile_id = auth.uid()
  limit 1;
$$;

create or replace function public.current_user_is_active()
returns boolean
language sql
stable
as $$
  select coalesce((
    select p.is_active
    from public.profiles p
    where p.id = auth.uid()
    limit 1
  ), false);
$$;

create or replace function public.can_current_medico_access_paciente(p_paciente_id uuid)
returns boolean
language sql
stable
as $$
  select exists (
    select 1
    from public.paciente_medico pm
    where pm.paciente_id = p_paciente_id
      and pm.medico_id = public.current_medico_id()
      and pm.is_active = true
  );
$$;

create or replace function public.can_current_user_manage_procedimento(p_procedimento_id uuid)
returns boolean
language sql
stable
as $$
  select
    case
      when public.is_admin() then true
      when public.is_medico() then exists (
        select 1
        from public.procedimentos_realizados pr
        where pr.id = p_procedimento_id
          and pr.medico_id = public.current_medico_id()
      )
      else false
    end;
$$;

create or replace function public.can_current_paciente_request_procedimento(p_procedimento_id uuid)
returns boolean
language sql
stable
as $$
  select exists (
    select 1
    from public.procedimentos_realizados pr
    where pr.id = p_procedimento_id
      and pr.paciente_id = public.current_paciente_id()
      and pr.is_active = true
  );
$$;

create or replace function public.has_current_paciente_active_access_to_procedimento(p_procedimento_id uuid)
returns boolean
language sql
stable
as $$
  select exists (
    select 1
    from public.acessos_imagem_procedimento a
    where a.procedimento_id = p_procedimento_id
      and a.paciente_id = public.current_paciente_id()
      and a.is_active = true
  );
$$;

drop policy if exists "paciente_medico_select_policy" on public.paciente_medico;
drop policy if exists "paciente_medico_insert_policy" on public.paciente_medico;
drop policy if exists "paciente_medico_update_policy" on public.paciente_medico;

create policy "profiles_select_policy"
on public.profiles
for select
to authenticated
using (
  public.current_user_is_active()
  and (
    public.is_admin()
    or id = public.current_profile_id()
  )
);

create policy "profiles_update_policy"
on public.profiles
for update
to authenticated
using (
  public.current_user_is_active()
  and (
    public.is_admin()
    or id = public.current_profile_id()
  )
)
with check (
  public.current_user_is_active()
  and (
    public.is_admin()
    or id = public.current_profile_id()
  )
);

create policy "medicos_select_policy"
on public.medicos
for select
to authenticated
using (
  public.current_user_is_active()
  and (
    public.is_admin()
    or profile_id = public.current_profile_id()
  )
);

create policy "medicos_insert_policy"
on public.medicos
for insert
to authenticated
with check (
  public.current_user_is_active()
  and public.is_admin()
);

create policy "medicos_update_policy"
on public.medicos
for update
to authenticated
using (
  public.current_user_is_active()
  and public.is_admin()
)
with check (
  public.current_user_is_active()
  and public.is_admin()
);

create policy "pacientes_select_policy"
on public.pacientes
for select
to authenticated
using (
  public.current_user_is_active()
  and (
    public.is_admin()
    or profile_id = public.current_profile_id()
    or (
      public.is_medico()
      and exists (
        select 1
        from public.paciente_medico pm
        where pm.paciente_id = pacientes.id
          and pm.medico_id = public.current_medico_id()
          and pm.is_active = true
      )
    )
  )
);

create policy "pacientes_insert_policy"
on public.pacientes
for insert
to authenticated
with check (
  public.current_user_is_active()
  and (
    public.is_admin()
    or public.is_medico()
  )
);

create policy "pacientes_update_policy"
on public.pacientes
for update
to authenticated
using (

import { createClient } from "npm:@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers":
    "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};

type AppRole = "admin" | "medico" | "paciente";

type ApprovePayload = {
  solicitacao_id: string;
  decision: "aprovada" | "reprovada";
  motivo_reprovacao?: string | null;
};

function getEnv() {
  const supabaseUrl = Deno.env.get("SUPABASE_URL");
  const supabaseAnonKey = Deno.env.get("SUPABASE_ANON_KEY");
  const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");

  if (!supabaseUrl || !supabaseAnonKey || !serviceRoleKey) {
    throw new Error("Missing required environment variables.");
  }

  return { supabaseUrl, supabaseAnonKey, serviceRoleKey };
}

function getBearerToken(req: Request): string {
  const authHeader = req.headers.get("Authorization") ?? "";
  if (!authHeader.startsWith("Bearer ")) {
    throw new Error("Missing or invalid Authorization header.");
  }
  return authHeader.replace("Bearer ", "").trim();
}

function createUserClient(accessToken: string) {
  const { supabaseUrl, supabaseAnonKey } = getEnv();

  return createClient(supabaseUrl, supabaseAnonKey, {
    global: {
      headers: {
        Authorization: `Bearer ${accessToken}`,
      },
    },
    auth: {
      persistSession: false,
      autoRefreshToken: false,
    },
  });
}

function createAdminClient() {
  const { supabaseUrl, serviceRoleKey } = getEnv();

  return createClient(supabaseUrl, serviceRoleKey, {
    auth: {
      persistSession: false,
      autoRefreshToken: false,
    },
  });
}

async function getCurrentUserContext(accessToken: string) {
  const userClient = createUserClient(accessToken);
  const adminClient = createAdminClient();

  const {
    data: { user },
    error: userError,
  } = await userClient.auth.getUser();

  if (userError || !user) {
    throw new Error("Unable to resolve authenticated user.");
  }

  const { data: profile, error: profileError } = await adminClient
    .from("profiles")
    .select("id, role, is_active")
    .eq("id", user.id)
    .single();

  if (profileError || !profile) {
    throw new Error("Profile not found.");
  }

  if (!profile.is_active) {
    throw new Error("User is inactive.");
  }

  return {
    profileId: profile.id as string,
    role: profile.role as AppRole,
    adminClient,
  };
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    if (req.method !== "POST") {
      return new Response(JSON.stringify({ error: "Method not allowed" }), {
        status: 405,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      });
    }

    const accessToken = getBearerToken(req);
    const { profileId, role, adminClient } = await getCurrentUserContext(accessToken);

    const body = (await req.json()) as ApprovePayload;
    const { solicitacao_id, decision, motivo_reprovacao } = body;

    if (!solicitacao_id || !decision) {
      return new Response(
        JSON.stringify({ error: "solicitacao_id and decision are required" }),
        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    if (decision !== "aprovada" && decision !== "reprovada") {
      return new Response(
        JSON.stringify({ error: "decision must be 'aprovada' or 'reprovada'" }),
        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    const { data: currentMedico } = await adminClient
      .from("medicos")
      .select("id, profile_id")
      .eq("profile_id", profileId)
      .maybeSingle();

    const { data: solicitacao, error: solicitacaoError } = await adminClient
      .from("solicitacoes_imagem")
      .select("id, procedimento_id, paciente_id, status")
      .eq("id", solicitacao_id)
      .single();

    if (solicitacaoError || !solicitacao) {
      return new Response(
        JSON.stringify({ error: "Solicitação não encontrada" }),
        { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    const { data: procedimento, error: procedimentoError } = await adminClient
      .from("procedimentos_realizados")
      .select("id, medico_id")
      .eq("id", solicitacao.procedimento_id)
      .single();

    if (procedimentoError || !procedimento) {
      return new Response(
        JSON.stringify({ error: "Procedimento não encontrado" }),
        { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    const isAdmin = role === "admin";
    const isResponsibleDoctor =
      role === "medico" &&
      currentMedico?.id != null &&
      procedimento.medico_id === currentMedico.id;

    if (!isAdmin && !isResponsibleDoctor) {
      return new Response(
        JSON.stringify({ error: "Sem permissão para decidir esta solicitação" }),
        { status: 403, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    const now = new Date().toISOString();

    const { error: updateSolicitacaoError } = await adminClient
      .from("solicitacoes_imagem")
      .update({
        status: decision,
        reviewed_at: now,
        reviewed_by: profileId,
        motivo_reprovacao: decision === "reprovada" ? (motivo_reprovacao ?? null) : null,
      })
      .eq("id", solicitacao_id);

    if (updateSolicitacaoError) throw updateSolicitacaoError;

    if (decision === "aprovada") {
      const { data: acessoExistente } = await adminClient
        .from("acessos_imagem_procedimento")
        .select("id")
        .eq("procedimento_id", solicitacao.procedimento_id)
        .eq("paciente_id", solicitacao.paciente_id)
        .maybeSingle();

      if (acessoExistente?.id) {
        const { error: reactivateError } = await adminClient
          .from("acessos_imagem_procedimento")
          .update({
            is_active: true,
            granted_by: profileId,
            granted_at: now,
            revoked_by: null,
            revoked_at: null,
          })
          .eq("id", acessoExistente.id);

        if (reactivateError) throw reactivateError;
      } else {
        const { error: createAccessError } = await adminClient
          .from("acessos_imagem_procedimento")
          .insert({
            procedimento_id: solicitacao.procedimento_id,
            paciente_id: solicitacao.paciente_id,
            granted_by: profileId,
            granted_at: now,
            is_active: true,
          });

        if (createAccessError) throw createAccessError;
      }
    }

    await adminClient.rpc("insert_audit_log_safe", {
      p_actor_profile_id: profileId,
      p_actor_role: role,
      p_action: decision === "aprovada" ? "APPROVE_IMAGE_REQUEST" : "REJECT_IMAGE_REQUEST",
      p_entity_type: "solicitacoes_imagem",
      p_entity_id: solicitacao.id,
      p_description:
        decision === "aprovada"
          ? "Solicitação de acesso aprovada"
          : "Solicitação de acesso reprovada",
      p_metadata: {
        procedimento_id: solicitacao.procedimento_id,
        paciente_id: solicitacao.paciente_id,
      },
    });

    return new Response(JSON.stringify({ success: true, status: decision }), {
      status: 200,
      headers: { ...corsHeaders, "Content-Type": "application/json" },
    });
  } catch (error) {
    return new Response(
      JSON.stringify({
        error: error instanceof Error ? error.message : "Unexpected error",
      }),
      {
        status: 500,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      },
    );
  }
});

function getBearerToken(req: Request): string {
  const authHeader = req.headers.get("Authorization") ?? "";
  if (!authHeader.startsWith("Bearer ")) {
    throw new Error("Missing or invalid Authorization header.");
  }
  return authHeader.replace("Bearer ", "").trim();
}

function createUserClient(accessToken: string) {
  const { supabaseUrl, supabaseAnonKey } = getEnv();

  return createClient(supabaseUrl, supabaseAnonKey, {
    global: {
      headers: {
        Authorization: `Bearer ${accessToken}`,
      },
    },
    auth: {
      persistSession: false,
      autoRefreshToken: false,
    },
  });
}

function createAdminClient() {
  const { supabaseUrl, serviceRoleKey } = getEnv();

  return createClient(supabaseUrl, serviceRoleKey, {
    auth: {
      persistSession: false,
      autoRefreshToken: false,
    },
  });
}

async function getCurrentUserContext(accessToken: string) {
  const userClient = createUserClient(accessToken);
  const adminClient = createAdminClient();

  const {
    data: { user },
    error: userError,
  } = await userClient.auth.getUser();

  if (userError || !user) {
    throw new Error("Unable to resolve authenticated user.");
  }

  const { data: profile, error: profileError } = await adminClient
    .from("profiles")
    .select("id, role, is_active")
    .eq("id", user.id)
    .single();

  if (profileError || !profile) {
    throw new Error("Profile not found.");
  }

  if (!profile.is_active) {
    throw new Error("User is inactive.");

  return {

    role: profile.role as AppRole,

  };
}

    const { profileId, role, adminClient } = await getCurrentUserContext(accessToken);

    if (!procedimento_id || !paciente_id) {
      return new Response(

        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    const { data: currentMedico } = await adminClient
      .from("medicos")
      .select("id, profile_id")

      .maybeSingle();

    const isResponsibleDoctor =
      role === "medico" &&
      currentMedico?.id != null &&
      procedimento.medico_id === currentMedico.id;

    if (!isAdmin && !isResponsibleDoctor) {

      .eq("paciente_id", paciente_id)
      .eq("is_active", true);

      p_actor_profile_id: profileId,
      p_actor_role: role,
      p_action: "REVOKE_IMAGE_ACCESS",
      p_entity_type: "acessos_imagem_procedimento",

      p_metadata: {
        procedimento_id,
        paciente_id,

      status: 200,

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    if (req.method !== "POST") {
      return new Response(JSON.stringify({ error: "Method not allowed" }), {

    }

    const accessToken = getBearerToken(req);
    const { role, profileId, adminClient } = await getCurrentUserContext(accessToken);
    const { imagesBucket } = getEnv();

    const body = (await req.json()) as SignedUrlPayload;
    const imagem_id = body.imagem_id;
    const expires_in = Math.min(Math.max(body.expires_in ?? 60, 30), 300);

    if (!imagem_id) {
      return new Response(
        JSON.stringify({ error: "imagem_id is required" }),
        { status: 400, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    const { data: imagem, error: imagemError } = await adminClient
      .from("imagens_procedimento")
      .select("id, procedimento_id, storage_path, is_active")
      .eq("id", imagem_id)
      .single();

    if (imagemError || !imagem || !imagem.is_active) {
      return new Response(
        JSON.stringify({ error: "Imagem não encontrada ou inativa" }),
        { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    const { data: procedimento, error: procedimentoError } = await adminClient
      .from("procedimentos_realizados")
      .select("id, paciente_id, medico_id")
      .eq("id", imagem.procedimento_id)
      .single();

    if (procedimentoError || !procedimento) {
      return new Response(
        JSON.stringify({ error: "Procedimento não encontrado" }),
        { status: 404, headers: { ...corsHeaders, "Content-Type": "application/json" } },
      );
    }

    let authorized = false;

    if (role === "admin") {
      authorized = true;
    } else if (role === "medico") {
      const { data: currentMedico } = await adminClient
        .from("medicos")
        .select("id")
        .eq("profile_id", profileId)
        .maybeSingle();

      authorized = currentMedico?.id === procedimento.medico_id;
    } else if (role === "paciente") {
      const { data: currentPaciente } = await adminClient
        .from("pacientes")
        .select("id")
        .eq("profile_id", profileId)
        .maybeSingle();

      if (currentPaciente?.id === procedimento.paciente_id) {
        const { data: acesso } = await adminClient
          .from("acessos_imagem_procedimento")
          .select("id")
          .eq("procedimento_id", procedimento.id)
          .eq("paciente_id", currentPaciente.id)
          .eq("is_active", true)
          .maybeSingle();

        authorized = !!acesso;

    }

    if (!authorized) {

        JSON.stringify({ error: "Sem permissão para acessar esta imagem" }),

      );

      .from(imagesBucket)
      .createSignedUrl(imagem.storage_path, expires_in);

    if (signedError || !signed?.signedUrl) {
      throw signedError ?? new Error("Unable to generate signed URL");
    }

    await adminClient.rpc("insert_audit_log_safe", {
      p_actor_profile_id: profileId,
      p_actor_role: role,
      p_action: "GENERATE_SIGNED_URL_FOR_IMAGE",
      p_entity_type: "imagens_procedimento",
      p_entity_id: imagem.id,
      p_description: "Signed URL gerada para acesso à imagem",
      p_metadata: {
        procedimento_id: procedimento.id,
        imagem_id: imagem.id,
        expires_in,
      },
    });

    return new Response(
      JSON.stringify({
        success: true,
        signed_url: signed.signedUrl,
        expires_in,
      }),
      {
        status: 200,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      },
    );
  } catch (error) {
    return new Response(
      JSON.stringify({
        error: error instanceof Error ? error.message : "Unexpected error",
      }),
      {
        status: 500,
        headers: { ...corsHeaders, "Content-Type": "application/json" },
      },

============================================================
OBSERVAÇÕES FINAIS DE ARQUITETURA
============================================================

1) NÃO tratar email como campo comum
------------------------------------------------------------
Email deve ser atualizado por fluxo controlado.
Não sair dando update em profiles.email pelo app como se fosse
um campo qualquer.

2) Senha
------------------------------------------------------------
Senha deve seguir:
- reset por email do Supabase Auth
ou
- fluxo controlado de update_user_credentials

3) Criptografia
------------------------------------------------------------
A chave única do sistema é aceitável para TCC,
mas é uma concessão de segurança.
Não tratar isso como arquitetura robusta de produção.

4) Bucket privado
------------------------------------------------------------
Bucket procedure-images deve continuar privado.
Paciente não deve ler diretamente do storage.
Paciente recebe signed URL curta via Edge Function.

5) Composição antes|depois
------------------------------------------------------------
A composição antes|depois deve ser feita no Flutter:
- baixar imagens autorizadas
- descriptografar localmente
- compor lado a lado
- compartilhar
- não salvar permanentemente

6) Upload
------------------------------------------------------------
Fluxo ideal no Flutter:
- capturar ou escolher imagem
- comprimir
- criptografar
- gerar imagem_id
- montar storage_path
- inserir metadado em imagens_procedimento
- subir binário no bucket
- atualizar status do procedimento se necessário

7) Próximo bloco lógico
------------------------------------------------------------
Depois deste backend, o próximo passo é:
- integração Flutter com upload criptografado
- download + descriptografia
- composição antes|depois
